-- Images table with PostGIS geometry
CREATE TABLE images (
    id SERIAL PRIMARY KEY,
    filename VARCHAR(255) NOT NULL,
    uploader_id INTEGER REFERENCES users(id),
    uploaded_at TIMESTAMP DEFAULT NOW(),
    -- PostGIS geometry instead of separate lat/lng
    location GEOMETRY(Point, 4326),  -- SRID 4326 = WGS84 (standard lat/lng)
    -- Administrative location information
    region VARCHAR(100),           -- Region (e.g., "Region IV-A (CALABARZON)")
    province VARCHAR(100),         -- Province (e.g., "Batangas")
    municipality VARCHAR(100),     -- Municipality/City (e.g., "Batangas City")
    barangay VARCHAR(100),         -- Barangay (e.g., "Poblacion")
    transect VARCHAR(50),          -- Transect identifier for research purposes
    site_name VARCHAR(255),        -- Optional site name for better identification
    quadrat_crop_path TEXT,
    original_image_path TEXT,
    -- Critical additions:
    total_pixels INTEGER,          -- Total pixels in analyzed area
    analyzed_area_px INTEGER,      -- Pixels actually analyzed (after masking invalid areas)
    analysis_confidence NUMERIC(3,2), -- How confident the ML model was
    updated_at TIMESTAMP,
    processing_status VARCHAR(20) DEFAULT 'pending', -- pending, processing, completed, failed
    -- Spatial index for performance
    CONSTRAINT enforce_srid_location CHECK (ST_SRID(location) = 4326)
);

-- Coral lifeforms reference table (unchanged, but good as is)
CREATE TABLE coral_lifeforms (
    id SERIAL PRIMARY KEY,
    class_name VARCHAR(100) NOT NULL UNIQUE,
    scientific_name VARCHAR(255),
    category VARCHAR(50), -- hard_coral, soft_coral, algae, etc.
    color_hex VARCHAR(7), -- For consistent visualization
    description TEXT
);

-- Enhanced segmentation results
CREATE TABLE segmentation_results (
    id SERIAL PRIMARY KEY,
    image_id INTEGER REFERENCES images(id) ON DELETE CASCADE,
    class_id INTEGER REFERENCES coral_lifeforms(id),
    instance_count INTEGER DEFAULT 0,
    area_px INTEGER NOT NULL,           -- Raw pixel count for this class
    coverage_percent NUMERIC(5,2),      -- (area_px / total_analyzed_area) * 100
    avg_confidence NUMERIC(3,2),        -- Average confidence for detections
    mask_path TEXT,                     -- Path to mask file
    created_at TIMESTAMP DEFAULT NOW()
);

-- Individual coral instance data
CREATE TABLE coral_instances (
    id SERIAL PRIMARY KEY,
    segmentation_id INTEGER REFERENCES segmentation_results(id),
    instance_number INTEGER,           -- Instance 1, 2, 3... within same class
    area_px INTEGER,                   -- Pixel area of this specific instance
    confidence NUMERIC(3,2),           -- Detection confidence
    bbox_coordinates JSONB             -- Bounding box coordinates if needed
);

-- Philippine Administrative Divisions reference table
-- This will store the hierarchical data for Philippines regions, provinces, municipalities, and barangays
CREATE TABLE philippine_locations (
    id SERIAL PRIMARY KEY,
    region_code VARCHAR(20),
    region_name VARCHAR(100) NOT NULL,
    province_code VARCHAR(20),
    province_name VARCHAR(100),
    municipality_code VARCHAR(20),
    municipality_name VARCHAR(100),
    barangay_code VARCHAR(20),
    barangay_name VARCHAR(100),
    location_type VARCHAR(20) NOT NULL, -- 'region', 'province', 'municipality', 'barangay'
    created_at TIMESTAMP DEFAULT NOW(),
    -- Ensure proper hierarchy
    CONSTRAINT chk_location_type CHECK (location_type IN ('region', 'province', 'municipality', 'barangay'))
);

-- Indexes for efficient cascading queries
CREATE INDEX idx_phil_locations_type ON philippine_locations (location_type);
CREATE INDEX idx_phil_locations_region ON philippine_locations (region_code, region_name);
CREATE INDEX idx_phil_locations_province ON philippine_locations (region_code, province_code, province_name);
CREATE INDEX idx_phil_locations_municipality ON philippine_locations (province_code, municipality_code, municipality_name);
